function cleanPdfItems(items){let out='',prev=null;for(const it of items){const str=(it.str||'').replace(/\u0000/g,'');if(!str)continue;const x=it.transform?.[4]??0,y=it.transform?.[5]??0,h=Math.abs(it.transform?.[3]||it.height||10),w=Number(it.width||0);if(prev){const tol=Math.max(2.5,Math.min(prev.h,h)*.38);if(Math.abs(y-prev.y)>tol){if(!out.endsWith('\n'))out+='\n';}else{const gap=x-(prev.x+prev.w);if(gap>Math.max(1.8,Math.min(prev.h,h)*.18)&&!/[\s\n]$/.test(out)&&!/^[,.;:!?%)\]}]/.test(str))out+=' ';}}out+=str;prev={x,y,h,w};}return out.replace(/[ \t]+\n/g,'\n').replace(/\n{3,}/g,'\n\n').replace(/[ \t]{2,}/g,' ').trim();}
async function extractPdf(file,progress){const buf=await file.arrayBuffer(),pdf=await pdfjsLib.getDocument({data:buf}).promise,out=[];for(let i=1;i<=pdf.numPages;i++){progress?.(i,pdf.numPages);const p=await pdf.getPage(i),c=await p.getTextContent({includeMarkedContent:true});out.push(cleanPdfItems(c.items));}return out;}

async function extractHwpx(file,progress){const zip=await JSZip.loadAsync(await file.arrayBuffer());const names=Object.keys(zip.files).filter(n=>/^Contents\/section\d+\.xml$/i.test(n)).sort((a,b)=>Number((a.match(/section(\d+)/i)||[])[1]||0)-Number((b.match(/section(\d+)/i)||[])[1]||0));if(!names.length)throw new Error('HWPX section XML not found');const out=[];for(let si=0;si<names.length;si++){progress?.(si+1,names.length);const xml=await zip.file(names[si]).async('string'),doc=new DOMParser().parseFromString(xml,'application/xml');if(doc.getElementsByTagName('parsererror').length)throw new Error('HWPX XML parse error');let bucket=[];const flush=()=>{const t=bucket.join('\n').replace(/\n{3,}/g,'\n\n').trim();if(t)out.push(t);bucket=[];};const paras=localEls(doc,'p');for(const p of paras){const br=/^(1|true)$/i.test(p.getAttribute('pageBreak')||'');if(br&&bucket.length)flush();const line=localEls(p,'t').map(t=>t.textContent||'').join('').replace(/\u0000/g,'').trimEnd();if(line.trim())bucket.push(line);}flush();}if(!out.length)throw new Error('HWPX text not found');return out;}
function docxNodeText(node){let out='';const walk=n=>{if(n.nodeType!==1)return;const ln=n.localName;if(ln==='t')out+=n.textContent||'';else if(ln==='tab')out+='\t';else if(ln==='br'){const typ=n.getAttribute('w:type')||n.getAttribute('type')||'';out+=typ==='page'?'\f':'\n';}else if(ln==='lastRenderedPageBreak')out+='\f';else for(const c of n.children)walk(c);};walk(node);return out;}
async function extractDocx(file,progress){const zip=await JSZip.loadAsync(await file.arrayBuffer()),entry=zip.file('word/document.xml');if(!entry)throw new Error('DOCX document.xml not found');progress?.(1,1);const xml=await entry.async('string'),doc=new DOMParser().parseFromString(xml,'application/xml');if(doc.getElementsByTagName('parsererror').length)throw new Error('DOCX XML parse error');const body=[...doc.getElementsByTagName('*')].find(x=>x.localName==='body');if(!body)throw new Error('DOCX body not found');const out=[];let bucket=[];const flush=()=>{const t=bucket.join('\n').replace(/\n{3,}/g,'\n\n').trim();if(t)out.push(t);bucket=[];};for(const child of body.children){if(child.localName==='p'){const before=localEls(child,'pageBreakBefore').length>0;if(before&&bucket.length)flush();const raw=docxNodeText(child),pieces=raw.split('\f');pieces.forEach((piece,i)=>{if(piece.trim())bucket.push(piece.trimEnd());if(i<pieces.length-1)flush();});}else if(child.localName==='tbl'){const rows=localEls(child,'tr');for(const r of rows){const cells=localEls(r,'tc').map(tc=>localEls(tc,'p').map(p=>docxNodeText(p).replace(/\f/g,' ')).join(' ').trim()).filter(Boolean);if(cells.length)bucket.push(cells.join('\t'));}}}flush();if(!out.length)throw new Error('DOCX text not found');return out;}
async function extractFile(file,progress){const t=getType(file);if(t==='pdf')return extractPdf(file,progress);if(t==='hwpx')return extractHwpx(file,progress);if(t==='docx')return extractDocx(file,progress);throw new Error('unsupported');}
