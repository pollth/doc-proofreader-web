// v3.7 patch: 교정/비교 결과 카테고리 필터 버튼
let singleFilterV37 = 'all';
let compareFilterV37 = 'all';

function ensureFilterBarsV37(){
  const proofSummary = $('#proofSummary');
  if(proofSummary && !$('#singleFilterBar')){
    const bar = document.createElement('div');
    bar.id = 'singleFilterBar';
    bar.className = 'result-filters';
    proofSummary.insertAdjacentElement('afterend', bar);
  }
  const compareSummary = $('#compareSummary');
  if(compareSummary && !$('#compareFilterBar')){
    const bar = document.createElement('div');
    bar.id = 'compareFilterBar';
    bar.className = 'result-filters';
    compareSummary.insertAdjacentElement('afterend', bar);
  }
}

function singleGroupV37(x){
  if(x.category === '문장부호 추천' || x.sub === '띄어쓰기') return 'punct';
  if(x.category === '검토 필요') return 'review';
  if(x.category === '확정 오류') return 'typo';
  return 'other';
}

function renderSingleFiltersV37(open){
  ensureFilterBarsV37();
  const counts = {all:open.length, typo:0, review:0, punct:0};
  open.forEach(x=>{ const g=singleGroupV37(x); if(counts[g] != null) counts[g]++; });
  const items = [
    ['all','전체'],['typo','오타'],['review','검토 필요'],['punct','문장부호/공백']
  ];
  $('#singleFilterBar').innerHTML = items.map(([k,label])=>
    `<button class="filter-btn ${singleFilterV37===k?'active':''}" data-sfilter="${k}">${label} <span>${counts[k]||0}</span></button>`
  ).join('');
}

function issueContextV37(x){
  const d = state.single;
  if(!d) return {left:'',right:''};
  const source = (d.originalUnits && d.originalUnits[x.unit-1] != null) ? d.originalUnits[x.unit-1] : (d.units[x.unit-1]||'');
  let idx = source.indexOf(x.before, Math.max(0,(x.index||0)-20));
  if(idx<0) idx = source.indexOf(x.before);
  if(idx<0) return {left:'',right:''};
  const leftPart = source.slice(0,idx).trim();
  const rightPart = source.slice(idx+(x.before||'').length).trim();
  return {
    left:leftPart ? (leftPart.split(/\s+/).pop()||'') : '',
    right:rightPart ? (rightPart.split(/\s+/)[0]||'') : ''
  };
}

renderIssues = function(){
  ensureFilterBarsV37();
  const open = state.issues.filter(x=>x.state==='open');
  renderSingleFiltersV37(open);
  const arr = singleFilterV37==='all' ? open : open.filter(x=>singleGroupV37(x)===singleFilterV37);
  const sum=$('#proofSummary'), list=$('#issueList');
  const c={}; open.forEach(x=>c[x.category]=(c[x.category]||0)+1);
  sum.textContent=`교정 결과 ${open.length}건 · 오타 ${open.filter(x=>singleGroupV37(x)==='typo').length} · 검토 ${open.filter(x=>singleGroupV37(x)==='review').length} · 문장부호/공백 ${open.filter(x=>singleGroupV37(x)==='punct').length}`;
  $('#applyCertainBtn').disabled=!open.some(x=>x.category==='확정 오류'&&x.auto&&x.after);
  if(!arr.length){list.innerHTML='<div class="empty-box">선택한 항목이 없습니다.</div>';return;}
  list.innerHTML=arr.map(x=>{
    const ctx=issueContextV37(x);
    const left=ctx.left?`<span class="ctx-word">${esc(ctx.left)}</span> `:'';
    const right=ctx.right?` <span class="ctx-word">${esc(ctx.right)}</span>`:'';
    const change=x.after?`<span class="before">${esc(x.before)}</span><span class="arrow">→</span><span class="after">${esc(x.after)}</span>`:`<span class="before no-strike">${esc(x.before)}</span>`;
    const label=singleGroupV37(x)==='typo'?'오타':singleGroupV37(x)==='review'?'검토 필요':'문장부호/공백';
    return `<article class="issue issue-compact"><div class="issue-top"><b>${x.unit}${unitLabel(state.single.type)} · ${esc(x.sub)}</b><span class="badge ${badgeClass(x.category)}">${label}</span></div><div class="issue-line compact-line">${left}${change}${right}</div><div class="issue-actions compact-actions"><button class="mini" data-i="${x.id}" data-act="loc">위치 보기</button>${x.after?`<button class="mini apply" data-i="${x.id}" data-act="apply">${x.category==='확정 오류'?'수정 적용':'추천 적용'}</button>`:''}<button class="mini" data-i="${x.id}" data-act="ignore">유지</button></div></article>`;
  }).join('');
};

function compareGroupV37(r){
  if(r.type==='추가') return 'add';
  if(r.type==='삭제') return 'del';
  if(r.type==='변경' && classifyDiff(r)==='문장부호/공백') return 'punct';
  if(r.type==='변경') return 'change';
  return 'other';
}

function renderCompareFiltersV37(rows){
  ensureFilterBarsV37();
  const counts={all:rows.length,change:0,add:0,del:0,punct:0};
  rows.forEach(r=>{const g=compareGroupV37(r); if(counts[g]!=null) counts[g]++;});
  const items=[['all','전체'],['change','변경'],['add','추가'],['del','삭제'],['punct','문장부호/공백']];
  $('#compareFilterBar').innerHTML=items.map(([k,label])=>`<button class="filter-btn ${compareFilterV37===k?'active':''}" data-cfilter="${k}">${label} <span>${counts[k]||0}</span></button>`).join('');
}

renderCompareRows = function(){
  ensureFilterBarsV37();
  const allRows=state.aligned.filter(r=>r.type!=='동일');
  renderCompareFiltersV37(allRows);
  const rows=compareFilterV37==='all'?allRows:allRows.filter(r=>compareGroupV37(r)===compareFilterV37);
  const c={변경:0,추가:0,삭제:0}; allRows.forEach(r=>c[r.type]=(c[r.type]||0)+1);
  $('#compareSummary').innerHTML=`차이 <b>${allRows.length}건</b> · 변경 ${c['변경']||0} · 추가 ${c['추가']||0} · 삭제 ${c['삭제']||0}`;
  if(!rows.length){$('#compareList').innerHTML='<div class="empty-box">선택한 항목이 없습니다.</div>';return;}
  $('#compareList').innerHTML=rows.map((r,k)=>{
    const idx=state.aligned.indexOf(r),kind=classifyDiff(r),cls=r.type==='추가'?'add':r.type==='삭제'?'del':'mod';
    return `<article class="compare-row" data-select-row="${idx}"><div class="compare-row-head"><b>${kind}</b><span class="badge ${cls}">${r.type}</span></div><div class="pair"><div class="pair-box"><div class="pair-label"><span>A 원본 · 기준</span><span>${r.a?r.a.unit+unitLabel(state.A.type):'—'}</span></div><textarea data-row="${idx}" data-side="A" ${r.a?'':'disabled'}>${esc(r.a?.text||'')}</textarea></div><div class="pair-box"><div class="pair-label"><span>B 작업본 · 확인대상</span><span>${r.b?r.b.unit+unitLabel(state.B.type):'—'}</span></div><textarea data-row="${idx}" data-side="B" ${r.b?'':'disabled'}>${esc(r.b?.text||'')}</textarea></div></div><div class="diff-view"><b>A:</b> ${diffHtml(r.a,r.b,'a')}<br><b>B:</b> ${diffHtml(r.a,r.b,'b')}</div><div class="compare-actions">${r.a?`<button class="mini" data-cact="locA" data-row="${idx}">A 원본에서 보기</button>`:''}${r.b?`<button class="mini" data-cact="locB" data-row="${idx}">B 작업본에서 보기</button>`:''}${r.a&&r.b?`<button class="mini apply" data-cact="AtoB" data-row="${idx}">A→B 수정</button><button class="mini apply" data-cact="BtoA" data-row="${idx}">B→A 수정</button><button class="mini" data-cact="saveA" data-row="${idx}">A 직접수정</button><button class="mini" data-cact="saveB" data-row="${idx}">B 직접수정</button>`:''}</div></article>`;
  }).join('');
};

ensureFilterBarsV37();
$('#singleFilterBar')?.addEventListener('click',e=>{const b=e.target.closest('[data-sfilter]');if(!b)return;singleFilterV37=b.dataset.sfilter;renderIssues();});
